<script>

const DB_NAME="FotoGeoDB", DB_VERSION=1, STORE="photos";

let db, currentOperator="", currentOperatorId="", currentPhotoBlob=null, currentPhotoName="";

let currentOperatorPhotos=[];


function openDB(){

 return new Promise((resolve,reject)=>{

  const r=indexedDB.open(DB_NAME,DB_VERSION);

  r.onupgradeneeded=()=>{if(!r.result.objectStoreNames.contains(STORE)){

    const s=r.result.createObjectStore(STORE,{keyPath:"id",autoIncrement:true});

    s.createIndex("operator","operator",{unique:false});

  }};

  r.onsuccess=()=>{db=r.result;resolve(db)}; r.onerror=()=>reject(r.error);

 });

}

function tx(mode="readonly"){return db.transaction(STORE,mode).objectStore(STORE)}

function allPhotos(){return new Promise((res,rej)=>{let r=tx().getAll();r.onsuccess=()=>res(r.result);r.onerror=()=>rej(r.error)})}

function addPhoto(o){return new Promise((res,rej)=>{let r=tx("readwrite").add(o);r.onsuccess=()=>res(r.result);r.onerror=()=>rej(r.error)})}

function deletePhoto(id){return new Promise((res,rej)=>{let r=tx("readwrite").delete(id);r.onsuccess=()=>res();r.onerror=()=>rej(r.error)})}


const $=id=>document.getElementById(id);

function esc(s){return String(s).replace(/[&<>"']/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[m]))}

function operatorKey(name){return name.trim().toLowerCase().replace(/\s+/g," ")}

function getOperators(photos){

 const map=new Map();

 photos.forEach(p=>{const k=operatorKey(p.operator); if(!map.has(k))map.set(k,{name:p.operator,count:0});map.get(k).count++});

 const saved=localStorage.getItem("fotogeo_operators");

 if(saved) try{JSON.parse(saved).forEach(o=>{if(!map.has(operatorKey(o.name)))map.set(operatorKey(o.name),{name:o.name,count:0})})}catch(e){}

 return [...map.values()].sort((a,b)=>a.name.localeCompare(b.name));

}

function saveOperators(list){localStorage.setItem("fotogeo_operators",JSON.stringify(list))}

async function renderOperators(){

 const photos=await allPhotos(), ops=getOperators(photos);

 if(!ops.length){$("operatorsList").innerHTML="Nessun operatore ancora registrato.";return}

 $("operatorsList").innerHTML=ops.map(o=>`

 <div class="operator">

  <div><div class="operator-name">👤 ${esc(o.name)}</div><div class="count">${o.count} foto</div></div>

  <button class="secondary" onclick="openGallery(${JSON.stringify(o.name)})">📁 Apri</button>

 </div>`).join("");

}

window.openGallery=async function(name){

 currentOperatorId=operatorKey(name); currentOperator=name;

 $("galleryTitle").textContent=`📁 ${name}`;

 $("gallerySection").classList.remove("hidden"); $("resultSection").classList.add("hidden");

 const photos=(await allPhotos()).filter(p=>operatorKey(p.operator)===currentOperatorId).sort((a,b)=>b.created-a.created);

 currentOperatorPhotos=photos;

 $("gallery").innerHTML=photos.length?photos.map(p=>`

  <div class="photo-item">

   <img src="${URL.createObjectURL(p.blob)}" alt="">

   <div><b>${esc(new Date(p.created).toLocaleString("it-IT"))}</b><br>${esc(p.lat.toFixed(6))}, ${esc(p.lon.toFixed(6))}

   <br><button class="danger" style="margin-top:7px;padding:7px 9px;font-size:12px" onclick="removePhoto(${p.id})">Elimina</button></div>

  </div>`).join(""):"<p class='muted'>Nessuna foto per questo operatore.</p>";

}

window.removePhoto=async function(id){if(confirm("Eliminare questa fotografia?")){await deletePhoto(id);openGallery(currentOperator);renderOperators()}}


$("backOperators").onclick=()=>{$("gallerySection").classList.add("hidden");renderOperators()};

$("saveOperator").onclick=()=>{

 const n=$("operator").value.trim(); if(!n){alert("Inserisci il nome dell'operatore.");return}

 currentOperator=n; localStorage.setItem("fotogeo_current_operator",n);

 const photosCache=[]; const ops=getOperators(photosCache);

 if(!ops.some(o=>operatorKey(o.name)===operatorKey(n))) {ops.push({name:n,count:0});saveOperators(ops)}

 $("gpsStatus").textContent=`👤 Operatore attivo: ${n}`;renderOperators();

};


$("takePhoto").onclick=()=>{

 const n=$("operator").value.trim()||localStorage.getItem("fotogeo_current_operator")||"";

 if(!n){alert("Inserisci prima il Nome operatore.");$("operator").focus();return}

 currentOperator=n; localStorage.setItem("fotogeo_current_operator",n);

 $("gpsStatus").textContent="📍 Avvio fotocamera e acquisizione GPS in background...";

 

 if(!navigator.geolocation){alert("Questo browser non supporta la geolocalizzazione.");return}


 window._gpsPromise = new Promise((resolve, reject) => {

   navigator.geolocation.getCurrentPosition(

     pos => resolve(pos.coords),

     err => reject(err),

     {enableHighAccuracy:true,timeout:15000,maximumAge:0}

   );

 });


 $("fileInput").click();

};


$("newPhoto").onclick=()=>$("takePhoto").click();


$("fileInput").onchange=async e=>{

  const file = e.target.files[0];

  if(!file) return;

  

  $("gpsStatus").textContent="📍 Attesa della posizione GPS...";

  

  try {

    const coords = await window._gpsPromise;

    window._gps = coords;

    await processImage(file, coords);

  } catch (err) {

    alert("Non è stato possibile ottenere la posizione GPS. Consenti la localizzazione, ricarica la pagina e riprova.");

    $("gpsStatus").textContent="⚠️ GPS non disponibile.";

  }

};


function pad(n){return String(n).padStart(2,"0")}

async function reverseGeocode(lat,lon){

 try{

  const r=await fetch(`https://nominatim.openstreetmap.org/reverse?format=jsonv2&lat=${encodeURIComponent(lat)}&lon=${encodeURIComponent(lon)}&zoom=18&addressdetails=1`,{headers:{"Accept-Language":"it"}});

  if(!r.ok)return "";

  const j=await r.json(); return j.display_name||"";

 }catch(e){return ""}

}

async function processImage(file,coords){

 const url=URL.createObjectURL(file), img=new Image();

 await new Promise((res,rej)=>{img.onload=res;img.onerror=rej;img.src=url});

 const c=document.createElement("canvas"), ctx=c.getContext("2d");

 c.width=img.naturalWidth;c.height=img.naturalHeight;ctx.drawImage(img,0,0);

 const now=new Date(), date=`${pad(now.getDate())}/${pad(now.getMonth()+1)}/${now.getFullYear()}`, time=`${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`;

 $("gpsStatus").textContent="📍 Recupero località...";

 const address=await reverseGeocode(coords.latitude,coords.longitude);

 const lines=[

  `OPERATORE: ${currentOperator}`,

  `${date} • ${time}`,

  `COORDINATE: ${coords.latitude.toFixed(6)}, ${coords.longitude.toFixed(6)}`,

  `ACCURATEZZA GPS: ±${Math.round(coords.accuracy||0)} m`

 ];

 if(address)lines.unshift(`📍 ${address}`);

 const scale=Math.max(1,img.naturalWidth/1200), fs=Math.max(22,Math.round(24*scale)), padX=Math.round(22*scale), lineH=Math.round(fs*1.45), boxH=padX+lineH*lines.length+padX;

 ctx.fillStyle="rgba(0,0,0,.70)";ctx.fillRect(0,c.height-boxH,c.width,boxH);

 ctx.font=`600 ${fs}px -apple-system,BlinkMacSystemFont,"Segoe UI",Arial,sans-serif`;ctx.fillStyle="#fff";ctx.textBaseline="top";

 lines.forEach((line,i)=>{

   let max=c.width-padX*2, text=line;

   while(ctx.measureText(text).width>max&&text.length>10)text=text.slice(0,-1);

   ctx.fillText(text,padX,c.height-boxH+padX+i*lineH);

 });

 const blob=await new Promise(r=>c.toBlob(r,"image/jpeg",.92));

 const name=`${date.replaceAll("/","-")}_${time.replaceAll(":","-")}.jpg`;

 const photo={operator:currentOperator,created:Date.now(),lat:coords.latitude,lon:coords.longitude,accuracy:coords.accuracy||0,address,blob,name};

 const id=await addPhoto(photo);photo.id=id;currentPhotoBlob=blob;currentPhotoName=name;

 $("preview").src=URL.createObjectURL(blob);$("downloadPhoto").href=URL.createObjectURL(blob);$("downloadPhoto").download=name;

 $("photoInfo").innerHTML=`<b>Operatore:</b> ${esc(currentOperator)}<br><b>Posizione:</b> ${coords.latitude.toFixed(6)}, ${coords.longitude.toFixed(6)}<br><b>Accuratezza:</b> ±${Math.round(coords.accuracy||0)} m${address?`<br><b>Località:</b> ${esc(address)}`:""}`;

 $("resultSection").classList.remove("hidden");$("gallerySection").classList.add("hidden");$("gpsStatus").textContent="✅ Foto geolocalizzata creata.";

 renderOperators();

}

$("sharePhoto").onclick=async()=>{

 if(!currentPhotoBlob)return;

 const file=new File([currentPhotoBlob],currentPhotoName,{type:"image/jpeg"});

 if(navigator.share&&navigator.canShare&&navigator.canShare({files:[file]})){

  try{await navigator.share({title:"Foto Geo",text:`Foto geolocalizzata - ${currentOperator}`,files:[file]})}catch(e){}

 }else{

  const a=document.createElement("a");a.href=URL.createObjectURL(currentPhotoBlob);a.download=currentPhotoName;a.click();

  alert("Il browser non supporta la condivisione diretta. La foto è stata scaricata; puoi allegarla a WhatsApp.");

 }

};

$("exportOperator").onclick=async()=>{

 if(!currentOperatorPhotos.length){alert("Nessuna foto da esportare.");return}

 alert("Per mantenere l'app completamente gratuita e senza server, l'esportazione crea un download per ogni foto. Potrai poi condividerle o archiviarle.");

 for(const p of currentOperatorPhotos){const a=document.createElement("a");a.href=URL.createObjectURL(p.blob);a.download=p.name;a.click();await new Promise(r=>setTimeout(r,250))}

};

(async()=>{

 await openDB();

 const saved=localStorage.getItem("fotogeo_current_operator");if(saved){$("operator").value=saved;currentOperator=saved}

 renderOperators();

})();

</script>
